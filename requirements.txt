# 安装: pip install -r requirements.txt

# 这是默认运行时集合：CLI + UVR 人声分离 + 规则字幕生成。
# ASR、VAD 和其他扩展仍需安装对应的可选依赖或使用 install.sh。
#
# 可选扩展:
#   pip install -e ".[faster-whisper]"       # ASR 语音识别 (faster-whisper)
#   pip install -e ".[funasr]"               # FunASR 中文语音识别 (WebUI 自动准备)
#   pip install -e ".[silero-vad]"           # Silero VAD (需 torch ~2GB)
#   pip install -e ".[llm]"                  # LLM 字幕优化
#   pip install -e ".[local-nlp]"            # 本地 NLP 语义合并 (CPU, ~120MB)
#   pip install -e ".[uvr]"                  # UVR 高品质人声分离 (BS-RoFormer，已包含在本文件中)
#   pip install -e ".[webui]"                # Web 图形界面
#   pip install -e ".[diarization]"          # 说话人分离
#   pip install -e ".[speaker-embedding]"    # 说话人嵌入 (SpeechBrain)
#   pip install -e ".[review-models]"        # Qwen/SED 模型下载与运行时基础依赖
#   pip install -e ".[qwen-runtime]"         # Qwen3-ASR runtime (Transformers 4.57.6)
#   pip install -e ".[whisperx]"             # WhisperX 词级时间戳增强 (强制音素对齐)
#   pip install -e ".[deepfilternet]"        # DeepFilterNet 语音增强 (降噪预处理)
#
# 一键部署:
#   bash install.sh                          # 交互式安装 (自动检测 GPU)
#   bash install.sh --cpu                    # 强制 CPU 模式
#   bash install.sh --gui                    # CLI + Web GUI
#   bash install.sh --all                    # 全量安装
#   pip install -r requirements-all.txt      # 全量依赖 (pip 方式)
#
# 说明:
#   - torch / torchaudio 已从核心依赖中移除，改为 [silero-vad] 可选附加
#   - UVR 是当前默认配置的分离引擎，因此作为默认运行时依赖安装
#   - faster-whisper 是 ASR 引擎，分开发布以避免强制安装 torch
# ============================================================
